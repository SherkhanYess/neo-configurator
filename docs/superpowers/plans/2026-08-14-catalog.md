# Neo Diamond Catalog — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Создать отдельное React-приложение-каталог с фильтром по огранке, сеткой украшений и страницей товара с 3D-вьювером и динамической ценой.

**Architecture:** Новый репо `neo-diamond-catalog` на Vite + React (viteSingleFile). Три экрана без роутера — state-based навигация как в конфигураторе. Переиспользует iJewel, priceCalc, и API-эндпоинты с основного конфигуратора.

**Tech Stack:** React 18, Vite, vite-plugin-singlefile, iJewel SDK, Netlify Functions (проксируем через con.neodiamond.kz или дублируем)

---

## Данные: все комбинации украшений

Каталог = декартово произведение:
- **10 огранок**: round, oval, pear, marquise, radiant, emerald, cushion, princess, heart, asscher
- **5 шинок**: Neo, Neo Luxe, Sirius, Sirius Luxe, Bezel
- **3 каста**: classic, halo, bezel

Итого: 10 × 5 × 3 = **150 карточек**. Каждая идентифицируется тройкой `{shape, shank, cast}`.

Россыпь (gem2) есть у: cast=halo (боковые камни у ореола). При cast=classic и cast=bezel gem2 отсутствует.

- [ ] **Шаг 3: src/lib/priceCalc.js**

```js
export function calcPrice(choices, prices) {
  if (!prices) return null;
  const base = prices.baseByShank?.[choices.shank] ?? 200000;
  const castAdd = choices.cast !== 'classic' ? (prices.casts?.[choices.cast] ?? 0) : 0;
  const carat = choices.carat ?? 1;
  const caratAdd = Math.max(0, carat - 1) * (prices.caratPrice ?? 200000);

  let gemPrice = prices.caratPrice ?? 200000;
  const gem1 = choices.gem1Label ?? '';
  if (gem1.toLowerCase().includes('fancy')) gemPrice += prices.fancyColorSurcharge ?? 100000;

  const purityAdd = choices.purity === '750' ? (prices.purity750surcharge ?? 20000) : 0;
  return base + castAdd + caratAdd * gemPrice / (prices.caratPrice ?? 200000) + purityAdd;
}

export function formatPrice(n) {
  if (!n) return '';
  return n.toLocaleString('ru-RU') + ' ₸';
}
```

```jsx
import React, { useCallback, useEffect, useRef, useState } from 'react';
import { useIjewel } from '../hooks/useIjewel.js';
import {
  CARAT_OPTIONS, GEM_COLORS, METAL_COLORS, METAL_PURITY, hasSideGems
} from '../lib/config.js';
import { calcPrice } from '../lib/priceCalc.js';
import { ViewerPanel } from '../components/ViewerPanel.jsx';
import { OptionRow, OptionPill } from '../components/OptionRow.jsx';
import { StickyBar } from '../components/StickyBar.jsx';

export function DetailScreen({ product, onBack, onBook }) {
  const ijewel = useIjewel();
  const [choices, setChoices] = useState({
    shape: product.shape.id,
    shank: product.shank.id,
    cast:  product.cast.id,
    carat: 1,
    purity: '585',
    gem1Label: 'D',
    gem2Label: null,
    metalLabel: 'Белое',
    castMetalLabel: 'Белое',
  });
  const [prices, setPrices] = useState(null);
  const restoredRef = useRef(false);

  useEffect(() => {
    fetch('/api/get-prices')
      .then((r) => r.ok ? r.json() : null)
      .then(setPrices)
      .catch(() => {});
  }, []);

  // Применить структуру украшения при готовности iJewel
  useEffect(() => {
    if (!ijewel.isReady || !ijewel.shankVariations.length) return;
    if (restoredRef.current) return;
    restoredRef.current = true;

    ijewel.applyHead(product.shape.ijewelTag, product.cast.ijewelTag);
    ijewel.applyShank(product.shank.id);
    ijewel.applyCarat(1);

    setTimeout(() => {
      ijewel.restoreFromLabels({
        gem1Label: 'D',
        gem2Label: null,
        metalLabel: 'Белое',
        castMetalLabel: 'Белое',
      });
    }, 1500);
  }, [ijewel.shankVariations]);

  function set(key, val) {
    setChoices((prev) => ({ ...prev, [key]: val }));
  }

  function applyMetal(label) {
    set('metalLabel', label);
    set('castMetalLabel', label);
    ijewel.applyMetalByLabel(label);
  }

  const handleInit = useCallback((el) => { ijewel.init(el); }, []);
  const price = calcPrice(choices, prices);
  const withSideGems = hasSideGems(product.cast.id);

  return (
    <div className="detail-screen">
      {/* Хедер */}
      <header className="detail-header">
        <button type="button" className="detail-back" onClick={onBack}>←</button>
        <span className="detail-title">{product.name}</span>
      </header>

      {/* 3D-вьювер */}
      <ViewerPanel onInit={handleInit} isReady={ijewel.isReady} />

      {/* Опции */}
      <div className="detail-options">

        <OptionRow label="Каратность">
          {CARAT_OPTIONS.map((v) => (
            <OptionPill
              key={v}
              label={`${v} кар`}
              selected={choices.carat === v}
              onClick={() => { set('carat', v); ijewel.applyCarat(v); }}
            />
          ))}
        </OptionRow>

        <OptionRow label="Цвет бриллианта">
          {GEM_COLORS.map((g) => (
            <OptionPill
              key={g.id}
              label={g.label}
              color={g.color}
              selected={choices.gem1Label === g.id}
              onClick={() => {
                set('gem1Label', g.id);
                ijewel.restoreFromLabels({ ...choices, gem1Label: g.id });
              }}
            />
          ))}
        </OptionRow>

        {withSideGems && (
          <OptionRow label="Цвет россыпи">
            {GEM_COLORS.map((g) => (
              <OptionPill
                key={g.id}
                label={g.label}
                color={g.color}
                selected={choices.gem2Label === g.id}
                onClick={() => {
                  set('gem2Label', g.id);
                  ijewel.restoreFromLabels({ ...choices, gem2Label: g.id });
                }}
              />
            ))}
          </OptionRow>
        )}

        <OptionRow label="Проба золота">
          {METAL_PURITY.map((p) => (
            <OptionPill
              key={p}
              label={p === '585' ? '585 · 14K' : '750 · 18K'}
              selected={choices.purity === p}
              onClick={() => set('purity', p)}
            />
          ))}
        </OptionRow>

        <OptionRow label="Цвет золота">
          {METAL_COLORS.map((m) => (
            <OptionPill
              key={m.id}
              label={m.label}
              color={m.color}
              selected={choices.metalLabel === m.label}
              onClick={() => applyMetal(m.label)}
            />
          ))}
        </OptionRow>

        <OptionRow label="Комбинированное золото">
          <OptionPill
            label="Шинка и каст одного цвета"
            selected={choices.metalLabel === choices.castMetalLabel}
            onClick={() => {
              set('castMetalLabel', choices.metalLabel);
              ijewel.restoreFromLabels({ ...choices, castMetalLabel: choices.metalLabel });
            }}
          />
          {METAL_COLORS.map((m) => (
            choices.metalLabel !== m.label && (
              <OptionPill
                key={m.id}
                label={`Каст — ${m.label}`}
                color={m.color}
                selected={choices.castMetalLabel === m.label}
                onClick={() => {
                  set('castMetalLabel', m.label);
                  ijewel.restoreFromLabels({ ...choices, castMetalLabel: m.label });
                }}
              />
            )
          ))}
        </OptionRow>

      </div>

      {/* Липкая нижняя панель */}
      <StickyBar price={price} onBook={() => onBook(choices)} />
    </div>
  );
}
```

```jsx
import React, { useState } from 'react';
import { FilterScreen } from './screens/FilterScreen.jsx';

export default function App() {

  const [selectedShapes, setSelectedShapes] = useState([]);
```

- [ ] **Шаг 2: LeadModal.jsx** — упрощённая версия, только имя + телефон + город + кнопка WA

```jsx
import React, { useState } from 'react';

const WA_BY_CITY = {
  'Алматы': '77766708505',
  'Астана': '77776908505',
  'Другой город': '77766708505',
};
const CITIES = ['Алматы', 'Астана', 'Другой город'];

export function LeadModal({ choices, onClose }) {
  const [name, setName] = useState('');
  const [phone, setPhone] = useState('');
  const [city, setCity] = useState('');

  function handleSubmit(e) {
    e.preventDefault();
    if (!name || !phone || !city) return;
    const wa = WA_BY_CITY[city];
    fetch('/api/submit-lead', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ name, phone: `+7${phone}`, city, config: choices }),
    }).catch(() => {});
    window.open(`https://wa.me/${wa}?text=${encodeURIComponent('Здравствуйте, хочу забронировать живой показ украшения!')}`, '_blank', 'noopener');
    onClose();
  }

  return (
    <div className="modal-overlay" onClick={(e) => e.target === e.currentTarget && onClose()}>
      <div className="modal">
        <button type="button" className="modal-close" onClick={onClose}>✕</button>
        <h3 className="modal-title">Забронировать живой показ</h3>
        <form onSubmit={handleSubmit}>
          <input className="modal-input" placeholder="Ваше имя" value={name} onChange={(e) => setName(e.target.value)} />
          <input className="modal-input" placeholder="Номер WhatsApp" value={phone} onChange={(e) => setPhone(e.target.value)} type="tel" />
          <select className="modal-input" value={city} onChange={(e) => setCity(e.target.value)}>
            <option value="">Выберите город</option>
            {CITIES.map((c) => <option key={c}>{c}</option>)}
          </select>
          <button type="submit" className="modal-submit">Перейти в WhatsApp</button>
        </form>
      </div>
    </div>
  );
}
```
